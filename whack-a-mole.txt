script.js:


let score = 0, time = 15, mole, timer;

const grid = document.getElementById("grid");

for (let i = 0; i < 9; i++) {
    let b = document.createElement("button");
    b.className = "hole";
    b.textContent = "•";

    b.onclick = () => {
        if (i === mole && time > 0) {
            score++;
            document.getElementById("score").textContent = score;
            showMole();
        }
    };

    grid.appendChild(b);
}

function showMole() {
    let holes = document.querySelectorAll(".hole");
    holes.forEach(b => b.textContent = "•");

    mole = Math.floor(Math.random() * 9);
    holes[mole].textContent = " X-X";
}

function startGame() {
    clearInterval(mole);
    clearInterval(timer);

    score = 0;
    time = 15;
    document.getElementById("score").textContent = score;
    document.getElementById("time").textContent = time;

    showMole();

    mole = setInterval(showMole, 700);

    timer = setInterval(() => {
        document.getElementById("time").textContent = --time;

        if (time === 0) {
            clearInterval(mole);
            clearInterval(timer);
            mole = -1;
            document.getElementById("message").textContent =
                "Game Over! Score: " + score;
        }
    }, 1000);
}

document.getElementById("start").onclick = startGame;
