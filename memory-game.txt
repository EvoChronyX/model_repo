<h1>Memory Game</h1>
<div id="game"></div>
<h3 id="msg">Find matching pairs!</h3>

<script>
let emojis=['🍎','🍌','🍇','🍒','🍎','🍌','🍇','🍒'];
emojis.sort(()=>Math.random()-0.5);
let first=null,lock=false,matches=0;
let game=document.getElementById('game');

emojis.forEach((emoji)=>{
let card=document.createElement('button');
card.className='card';
card.textContent='?';
game.appendChild(card);

card.onclick=()=>{
if(lock||card.textContent!=='?'&& !card.classList.contains('wrong'))return;
if(card.classList.contains('matched'))return;

card.textContent=emoji;

if(!first){
first=card;
return;
}

if(first===card)return;

if(first.textContent===card.textContent){
first.classList.add('matched');
card.classList.add('matched');
first.disabled=true;
card.disabled=true;
matches++;
first=null;

if(matches===4)
document.getElementById('msg').textContent='You Win!';
}else{
lock=true;
let previous=first;
first=null;
setTimeout(()=>{
previous.textContent='?';
card.textContent='?';
lock=false;
},800);
}
};
});
</script>
