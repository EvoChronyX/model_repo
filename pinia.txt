import { defineStore } from 'pinia'
import axios from 'axios'

export const useEmployeeStore = defineStore('employee', { 
    state: () => ({
        employees: []
    }),

    actions: {
        
        async fetchEmplyees()  {
            const res= await axios.get('localhost:8080/api/student')
       
          this.employees = res.data
           },

           async addemployee(employee) {

            const res =  await axios.post('localhost:8000/api/students', employee)
            this.employee.push(res.data)
           }
        }
 })

import { reactive } from 'vue'
import { useEmployeeStore } from '../stores/employeeStore'

const store = useEmployeeStore()


const employee =   reactive({
    name: '',
    email: email ''
})



async function submit() {

  await store.addEmployee({ ...employee })

  employee.name = ''
  employee.department = ''
  employee.email = ''
}


<template>

  <form @submit.prevent="submit">
    <input v-model="employee.name" placeholder="Name" required />
    <input v-model="employee.department" placeholder="Department" required />
    <input v-model="employee.email" type="email" placeholder="Email" required />
    <button>Add</button>
  </form>

</template>
