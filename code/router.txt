src / components / Employeeview.vue


<script setup>
import { ref, onMounted } from 'vue'
import axios from 'axios'

const employees = ref([])

onMounted(async () => {
  const response = await axios.get(
    'http://localhost:8080/api/employees'
  )
  employees.value = response.data
})
</script>

<template>
  <h2>Employee List</h2>

  <table border="1" cellpadding="8">
    <tr>
      <th>Name</th>
      <th>Department</th>
      <th>Email</th>
    </tr>

    <tr v-for="employee in employees" :key="employee.id">
      <td>{{ employee.name }}</td>
      <td>{{ employee.department }}</td>
      <td>{{ employee.email }}</td>
    </tr>
  </table>
</template>


AddEmployee.vue:

<script setup>
import { reactive } from 'vue'
import axios from 'axios'

const employee = reactive({
  name: '',
  department: '',
  email: ''
})

async function addEmployee() {
  await axios.post(
    'http://localhost:8080/api/employees',
    employee
  )

  alert('Employee added successfully!')

  employee.name = ''
  employee.department = ''
  employee.email = ''
}
</script>

<template>
  <h2>Add Employee</h2>

  <form @submit.prevent="addEmployee">
    <input v-model="employee.name" placeholder="Name" required />
    <br /><br />

    <input v-model="employee.department" placeholder="Department" required />
    <br /><br />

    <input v-model="employee.email" type="email" placeholder="Email" required />
    <br /><br />

    <button type="submit">Add Employee</button>
  </form>
</template>
